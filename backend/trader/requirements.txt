# Crypto Trading System — trader. Python 3.12+.
#
# EXACT PINS, deliberately.
#
# These were floating or wholly unconstrained: `scikit-learn`, `lightgbm`,
# `scipy` and `joblib` carried no version at all, and `coinbase-advanced-py` no
# constraint whatsoever — so `pip install` at image-build time pulled whatever was
# newest that day, in exactly the packages that deserialize the promoted model
# (`joblib`, `lightgbm`) and sign live orders (`cryptography`). A build that
# cannot be reproduced cannot be audited, and a model artifact is only loadable by
# the library version that wrote it.
#
# The versions below are the ones the 345-test suite passes against. Move them one
# at a time, on purpose. pandas 3.0 in particular is a live constraint: it is why
# `core/datastore.py` binds query parameters as `datetime` rather than
# `pd.Timestamp`, which under 3.0 stopped being implicitly convertible and made
# every `--start`/`--end` run raise.

# ---- data -------------------------------------------------------------------
numpy==2.3.5
pandas==3.0.0
pyarrow==25.0.1
duckdb==1.5.5

# ---- modelling --------------------------------------------------------------
# `joblib` writes and reads models/forecast.joblib; `lightgbm` owns the booster
# inside it. A mismatch here is an artifact that will not load, or worse, loads
# and scores differently.
scipy==1.17.0
scikit-learn==1.8.0
lightgbm==4.6.0
joblib==1.5.3

# ---- venues -----------------------------------------------------------------
aiohttp==3.13.3
coinbase-advanced-py==1.8.2
# Kalshi authentication is RSA-PSS over SHA-256, not an HMAC secret, so this is a
# hard requirement rather than a transitive one: kalshi_client imports it at
# signing time and the container fails on the first live cycle without it.
cryptography==46.0.3

# ---- serving store ---------------------------------------------------------
sqlalchemy==2.0.43
psycopg2-binary==2.9.11

# ---- optional ---------------------------------------------------------------
# The in-memory queue is the default; RedisQueue is implemented and never
# instantiated (pipeline.py hardwires InMemoryQueue).
redis==5.2.1

# ---- testing ---------------------------------------------------------------
pytest==8.3.3
# pytest.ini sets `-n auto`, so this is required rather than optional: without it
# pytest exits on an unrecognised argument.
pytest-xdist==3.8.0
pytest-asyncio==1.3.0

# ---- development ------------------------------------------------------------
black==26.1.0
isort==7.0.0
mypy==1.19.1
ruff==0.16.4
